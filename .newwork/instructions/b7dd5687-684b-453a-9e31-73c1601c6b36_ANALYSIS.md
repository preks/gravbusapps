### Step 2: Create CSV Parsing and Validation Utilities

**File**: `crm/customers/apps/customer-relations/views/contact-view/utils/csvUtils.ts`

```typescript
export interface CsvParseResult {
  data: Array<Record<string, string>>;
  errors: Array<{ row: number; field: string; message: string }>;
  warnings: Array<{ row: number; field: string; message: string }>;
}

export const parseCsv = (csvText: string): CsvParseResult => {
  const lines = csvText.split('\n').filter(line => line.trim());
  
  if (lines.length === 0) {
    return { data: [], errors: [{ row: 0, field: 'file', message: 'CSV file is empty' }], warnings: [] };
  }

  // Parse headers
  const headers = lines[0].split(',').map(h => h.trim());
  
  // Parse data rows
  const data: Array<Record<string, string>> = [];
  const errors: Array<{ row: number; field: string; message: string }> = [];
  
  for (let i = 1; i < lines.length; i++) {
    const values = lines[i].split(',').map(v => v.trim());
    const row: Record<string, string> = {};
    
    headers.forEach((header, index) => {
      row[header] = values[index] || '';
    });
    
    data.push(row);
  }

  return { data, errors, warnings: [] };
};

export const validateCsvHeaders = (headers: string[]): string[] => {
  const requiredHeaders = ['firstName', 'lastName', 'email'];
  const missing = requiredHeaders.filter(h => !headers.includes(h));
  return missing;
};
```

**Rationale:** Similar pattern to `src/business-apps/platform/apps/catalog-manager/utils/csvUtils.ts`.

### Step 3: Create Contact Validation Utility

**File**: `crm/customers/apps/customer-relations/views/contact-view/utils/contactValidation.ts`

```typescript
export interface ContactValidationError {
  field: string;
  message: string;
}

export const validateContactRow = (
  contact: Record<string, string>,
  rowNumber: number
): ContactValidationError[] => {
  const errors: ContactValidationError[] = [];

  // Required field validation
  if (!contact.firstName && !contact.lastName) {
    errors.push({
      field: 'name',
      message: `Row ${rowNumber}: At least firstName or lastName is required`
    });
  }

  // Email validation
  if (!contact.email) {
    errors.push({
      field: 'email',
      message: `Row ${rowNumber}: Email is required`
    });
  } else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(contact.email)) {
    errors.push({
      field: 'email',
      message: `Row ${rowNumber}: Invalid email format`
    });
  }

  // Phone validation (optional but format check if provided)
  if (contact.phone && !/^[\d\s\+\-\(\)]+$/.test(contact.phone)) {
    errors.push({
      field: 'phone',
      message: `Row ${rowNumber}: Invalid phone format`
    });
  }

  return errors;
};
```

**Rationale:** Validation logic extracted from `src/business-apps/crm/common/forms/contact-form/components/ContactFormComponent.tsx`.

### Step 5: Create Import Hook

**File**: `crm/customers/apps/customer-relations/views/contact-view/hooks/useContactImport.ts`

```typescript
import { useState, useCallback } from 'react';
import { BusinessAppContextProps } from '@/common/BusinessAppContext';
import { useRestClient } from '@/common/network/rest/useRestClient';
import { env } from '@/common/env/EnvUtils';
import { parseCsv, validateCsvHeaders } from '../utils/csvUtils';
import { validateContactRow } from '../utils/contactValidation';
import { ImportResult, ImportProgress, ImportedContact } from '../types/importTypes';

export const useContactImport = (context: BusinessAppContextProps) => {
  const [progress, setProgress] = useState<ImportProgress>({
    status: 'idle',
    progress: 0,
    message: ''
  });
  const [importResult, setImportResult] = useState<ImportResult | null>(null);

  const restClient = useRestClient(env('NEXT_PUBLIC_CRM_API_ENDPOINT') || '');

  const parseAndValidateCsv = useCallback((file: File): Promise<ImportedContact[]> => {
    return new Promise((resolve, reject) => {
      const reader = new FileReader();
      
      reader.onload = (e) => {
        try {
          setProgress({ status: 'parsing', progress: 25, message: 'Parsing CSV file...' });
          
          const csvText = e.target?.result as string;
          const { data, errors: parseErrors } = parseCsv(csvText);
          
          if (parseErrors.length > 0) {
            reject(new Error(`CSV parsing failed: ${parseErrors[0].message}`));
            return;
          }

          // Validate headers
          const headers = Object.keys(data[0] || {});
          const missingHeaders = validateCsvHeaders(headers);
          if (missingHeaders.length > 0) {
            reject(new Error(`Missing required columns: ${missingHeaders.join(', ')}`));
            return;
          }

          setProgress({ status: 'validating', progress: 50, message: 'Validating contacts...' });

          // Validate each row
          const contacts: ImportedContact[] = data.map((row, index) => {
            const validationErrors = validateContactRow(row, index + 2); // +2 for 1-based + header row
            
            return {
              ...row,
              isValid: validationErrors.length === 0,
              errors: validationErrors.map(e => e.message)
            };
          });

          resolve(contacts);
        } catch (error) {
          reject(error);
        }
      };

      reader.onerror = () => reject(new Error('Failed to read CSV file'));
      reader.readAsText(file);
    });
  }, []);

  const importContacts = useCallback(async (contacts: ImportedContact[]): Promise<ImportResult> => {
    setProgress({ status: 'uploading', progress: 75, message: 'Uploading contacts...' });

    try {
      // Filter out invalid contacts
      const validContacts = contacts.filter(c => c.isValid);
      const invalidContacts = contacts.filter(c => !c.isValid);

      // Prepare payload
      const payload = {
        contacts: validContacts.map(c => ({
          sourceApp: 'CRM',
          firstName: c.firstName,
          lastName: c.lastName,
          email: c.email,
          phone: c.phone,
          jobTitle: c.jobTitle,
          department: c.department,
          contactType: c.contactType || 'CUSTOMER',
          linkedinUrl: c.linkedinUrl,
          xingUrl: c.xingUrl,
          bio: c.bio,
          companyName: c.companyName,
          recordState: 'ACTIVE'
        }))
      };

      // Call backend API for bulk import
      const response = await restClient.post<ImportResult>(
        '/contacts/bulk-import',
        payload,
        {
          headers: {
            'Authorization': `Bearer ${context.accessToken}`,
            'Content-Type': 'application/json'
          }
        }
      );

      const result: ImportResult = {
        total: contacts.length,
        successful: response.successful || validContacts.length,
        failed: response.failed || invalidContacts.length,
        errors: invalidContacts.map((contact, index) => ({
          row: index + 2,
          contact,
          error: contact.errors.join(', ')
        }))
      };

      setImportResult(result);
      setProgress({ 
        status: 'completed', 
        progress: 100, 
        message: `Import completed: ${result.successful} successful, ${result.failed} failed` 
      });

      return result;
    } catch (error) {
      setProgress({ 
        status: 'error', 
        progress: 0, 
        message: error instanceof Error ? error.message : 'Import failed' 
      });
      throw error;
    }
  }, [context.accessToken, restClient]);

  const processImport = useCallback(async (file: File): Promise<ImportResult> => {
    try {
      const contacts = await parseAndValidateCsv(file);
      return await importContacts(contacts);
    } catch (error) {
      setProgress({ 
        status: 'error', 
        progress: 0, 
        message: error instanceof Error ? error.message : 'Import failed' 
      });
      throw error;
    }
  }, [parseAndValidateCsv, importContacts]);

  const resetImport = useCallback(() => {
    setProgress({ status: 'idle', progress: 0, message: '' });
    setImportResult(null);
  }, []);

  return {
    progress,
    importResult,
    processImport,
    resetImport
  };
};
```

**Rationale:** Combines patterns from:
- `src/common/data/catalog/hooks/useImportCatalogEntries.ts` (import logic)
- `src/business-apps/crm/investments/forms/investment-report-upload-form/hooks/useInvestmentFileUpload.ts` (file processing)

```typescript
interface ContactImportDialogProps extends BusinessAppContextProps {
  open: boolean;
  onClose: () => void;
  onSuccess?: () => void;
}

export const ContactImportDialog: React.FC<ContactImportDialogProps> = ({
  open,
  onClose,
  onSuccess,
  ...context
}) => {
  const [selectedFile, setSelectedFile] = useState<File | null>(null);
  const [dragActive, setDragActive] = useState(false);
  const fileInputRef = useRef<HTMLInputElement>(null);

  const { downloadTemplate } = useContactCsvTemplate();
  const { progress, importResult, processImport, resetImport } = useContactImport(context);

  const handleFileSelect = (file: File) => {
    if (file.type !== 'text/csv' && !file.name.endsWith('.csv')) {
      alert('Please select a CSV file');
      return;

    if (file.size > 5 * 1024 * 1024) { // 5MB limit

    e.preventDefault();
    e.stopPropagation();
    setDragActive(false);

      handleFileSelect(file);
    }

    e.preventDefault();
    e.stopPropagation();
    setDragActive(true);

    e.preventDefault();
    e.stopPropagation();
    setDragActive(false);
  };

    if (!selectedFile) return;

    try {
      const result = await processImport(selectedFile);
      
      if (result.successful > 0) {
        setTimeout(() => {
          onSuccess?.();
        }, 2000);
      }
    } catch (error) {
      console.error('Import error:', error);
    }
  };

  const handleClose = () => {
    setSelectedFile(null);
    resetImport();
    onClose();
  };

  return (
    <Dialog open={open} onOpenChange={handleClose}>
      <DialogContent className="gba:max-w-3xl" data-testid="contact-import-dialog">
        <DialogHeader>
          <DialogTitle>Import Contacts from CSV</DialogTitle>
          <DialogDescription>

          </DialogDescription>
        </DialogHeader>

        <div className="gba:space-y-4">
          {/* Download Template Button */}
          <div className="gba:flex gba:justify-end">
            <Button
              variant="ghost"
              size="sm"
              onClick={downloadTemplate}
              className="gba:flex gba:items-center gba:gap-2"
              data-testid="download-template-button"
            >
              <DownloadSimple className="gba:h-4 gba:w-4" />
              Download CSV Template
            </Button>
          </div>

          {/* File Upload Area */}
          {!selectedFile && progress.status === 'idle' && (
            <Card
              className={`gba:p-8 gba:border-2 gba:border-dashed gba:cursor-pointer gba:transition-colors ${
                dragActive ? 'gba:border-primary gba:bg-primary/5' : 'gba:border-muted'
              }`}
              onDrop={handleDrop}
              onDragOver={handleDragOver}
              onDragLeave={handleDragLeave}

              data-testid="file-upload-area"
            >
              <div className="gba:flex gba:flex-col gba:items-center gba:gap-3 gba:text-center">
                <Upload className="gba:h-12 gba:w-12 gba:text-muted-foreground" />
                <div>
                  <p className="gba:text-sm gba:font-medium">
                    Drag and drop your CSV file here, or click to browse
                  </p>
                  <p className="gba:text-xs gba:text-muted-foreground gba:mt-1">
                    Maximum file size: 5MB
                  </p>
                </div>
              </div>
              <input
                ref={fileInputRef}
                type="file"
                accept=".csv"
                className="gba:hidden"
                onChange={(e) => {
                  const file = e.target.files?.[0];
                  if (file) handleFileSelect(file);
                }}
                data-testid="file-input"
              />
            </Card>
          )}

          {/* Selected File Info */}
          {selectedFile && progress.status === 'idle' && (
            <Card className="gba:p-4">
              <div className="gba:flex gba:items-center gba:justify-between">
                <div className="gba:flex gba:items-center gba:gap-3">
                  <Upload className="gba:h-8 gba:w-8 gba:text-primary" />
                  <div>
                    <p className="gba:text-sm gba:font-medium">{selectedFile.name}</p>
                    <p className="gba:text-xs gba:text-muted-foreground">
                      {(selectedFile.size / 1024).toFixed(2)} KB
                    </p>
                  </div>
                </div>
                <Button
                  variant="ghost"
                  size="sm"
                  onClick={() => setSelectedFile(null)}
                  data-testid="remove-file-button"
                >
                  <X className="gba:h-4 gba:w-4" />
                </Button>
              </div>
            </Card>
          )}

          {/* Progress Indicator */}
          {progress.status !== 'idle' && progress.status !== 'completed' && (
            <Alert data-testid="import-progress-alert">
              <Spinner className="gba:h-4 gba:w-4" />
              <AlertDescription>
                {progress.message} ({progress.progress}%)
              </AlertDescription>
            </Alert>
          )}

          {/* Import Results */}
          {progress.status === 'completed' && importResult && (
            <Alert variant="default" data-testid="import-results-alert">
              <AlertDescription>
                <div className="gba:space-y-2">
                  <p className="gba:font-medium">Import Completed</p>
                  <div className="gba:flex gba:gap-2">
                    <Badge variant="success">
                      {importResult.successful} Successful
                    </Badge>
                    {importResult.failed > 0 && (
                      <Badge variant="destructive">
                        {importResult.failed} Failed
                      </Badge>
                    )}
                  </div>
                </div>
              </AlertDescription>
            </Alert>
          )}

          {/* Error Alert */}
          {progress.status === 'error' && (
            <Alert variant="destructive" data-testid="import-error-alert">
              <AlertDescription>{progress.message}</AlertDescription>
            </Alert>
          )}
        </div>

        <DialogFooter>
          <Button variant="outline" onClick={handleClose} data-testid="cancel-import-button">
            Cancel
          </Button>
          <Button
            onClick={handleImport}
            disabled={!selectedFile || progress.status !== 'idle'}
            data-testid="confirm-import-button"
          >
            {progress.status === 'uploading' ? (
              <>
                <Spinner className="gba:h-4 gba:w-4 gba:mr-2" />
                Importing...
              </>
            ) : (
              'Import Contacts'
            )}
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  );
};
```

**Rationale:** Combines patterns from:
- `src/business-apps/crm/investments/forms/investment-report-upload-form/InvestmentReportUploadForm.tsx` (file upload UI)
- `src/business-apps/crm/investments/apps/investment-report-portal/components/ReportUploadModal.tsx` (dialog structure)

## 🚨 Backend API Requirements

**Endpoint**: `POST /api/crm/contacts/bulk-import`

**Request**:
```json
{
  "contacts": [
    {
      "sourceApp": "CRM",
      "firstName": "John",
      "lastName": "Doe",
      "email": "john.doe@example.com",
      "phone": "+49 123 456789",
      "jobTitle": "Sales Manager",
      "department": "Sales",
      "contactType": "CUSTOMER",
      "linkedinUrl": "https://linkedin.com/in/johndoe",
      "xingUrl": "https://xing.com/profile/johndoe",
      "bio": "Experienced sales professional",
      "companyName": "ACME Corp",
      "recordState": "ACTIVE"
    }
  ]
}
```

**Response**:
```json
{
  "successful": 10,
  "failed": 2,
  "errors": [
    {
      "row": 3,
      "email": "invalid@",
      "error": "Invalid email format"
    },
    {
      "row": 5,
      "email": "duplicate@example.com",
      "error": "Contact already exists"
    }
  ]
}
```

**Error Handling**:
- Duplicate emails: Skip and report in errors array
- Invalid data: Skip and report in errors array
- Database errors: Return 500 with error message
