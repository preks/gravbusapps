1. **Market Data API Integration**
   - External API client for real-time stock prices, bond yields, market indices
   - Support for multiple data providers (Alpha Vantage, Yahoo Finance, IEX Cloud)
   - API key management and rate limiting
   - Data caching and refresh strategies

2. **Live Portfolio Valuation**
   - Real-time calculation of portfolio values based on current market prices
   - Automatic updates for equity investments (stocks)
   - Market index tracking for liquid funds
   - Currency conversion for international investments

3. **Real-Time Monitoring (Optional)**
   - WebSocket connection for live price feeds
   - Price alerts and notifications
   - Threshold-based alerts (e.g., price drops 5%)

4. **Performance Analysis Enhancements**
   - Compare portfolio performance against market benchmarks (S&P 500, DAX, etc.)
   - Intraday performance tracking
   - Market correlation analysis

### Phase 1: Market Data API Client (Foundation)

#### Step 1.1: Create MarketDataClient

**File:** `src/common/network/market-data/MarketDataClient.ts`

**Pattern:** Based on `src/common/network/investment-reports/InvestmentReportClient.ts`

```typescript
import { env } from '@/common/env/EnvUtils';
import { setAccessTokenToAuthorizationHeader } from '../HeaderUtils';
import { RestClient } from '../rest/RestClient';

export interface StockQuote {
  symbol: string;
  price: number;
  previousClose: number;
  open: number;
  dayHigh: number;
  dayLow: number;
  volume: number;
  marketCap: number;
  change: number;
  changePercent: number;
  lastUpdated: string;
}

export interface BatchQuotesResponse {
  quotes: Record<string, StockQuote>;
  errors?: Record<string, string>;
}

export class MarketDataClient extends RestClient {
  constructor(accessToken: string) {
    // Use Alpha Vantage or Yahoo Finance API
    const baseUrl = env('NEXT_PUBLIC_MARKET_DATA_API_URL') || 'https://www.alphavantage.co';
    const apiKey = env('NEXT_PUBLIC_MARKET_DATA_API_KEY') || '';
    
    super(baseUrl, {
      'Authorization': setAccessTokenToAuthorizationHeader(accessToken),
      'X-API-Key': apiKey
    });
  }

  async getStockQuote(ticker: string): Promise<StockQuote> {
    const response = await this.get<any>(`/query`, {
      params: {
        function: 'GLOBAL_QUOTE',
        symbol: ticker,
        apikey: env('NEXT_PUBLIC_MARKET_DATA_API_KEY')
      }
    });
    
    return this.transformAlphaVantageQuote(response);
  }

  async getBatchQuotes(tickers: string[]): Promise<BatchQuotesResponse> {
    // Batch fetch multiple stocks
    const promises = tickers.map(ticker => 
      this.getStockQuote(ticker).catch(error => ({ error, ticker }))
    );
    
    const results = await Promise.allSettled(promises);
    
    const quotes: Record<string, StockQuote> = {};
    const errors: Record<string, string> = {};
    
    results.forEach((result, index) => {
      if (result.status === 'fulfilled') {
        quotes[tickers[index]] = result.value;
      } else {
        errors[tickers[index]] = result.reason.message;
      }
    });
    
    return { quotes, errors };
  }

  private transformAlphaVantageQuote(response: any): StockQuote {
    const quote = response['Global Quote'];
    return {
      symbol: quote['01. symbol'],
      price: parseFloat(quote['05. price']),
      previousClose: parseFloat(quote['08. previous close']),
      open: parseFloat(quote['02. open']),
      dayHigh: parseFloat(quote['03. high']),
      dayLow: parseFloat(quote['04. low']),
      volume: parseInt(quote['06. volume']),
      marketCap: 0, // Not provided by Alpha Vantage
      change: parseFloat(quote['09. change']),
      changePercent: parseFloat(quote['10. change percent'].replace('%', '')),
      lastUpdated: quote['07. latest trading day']
    };
  }
}
```

```typescript
import { useState, useCallback, useEffect } from 'react';
import { BusinessAppContextProps } from '@/common/BusinessAppContext';
import { useMarketDataClient } from '@/common/network/market-data/useMarketDataClient';
import { useGetInvestmentsWithTickers } from '../queries/MarketDataQueries';

export interface EnhancedInvestmentWithMarketData {
  id: string;
  code: string;
  name: string;
  investmentType: string;
  totalInvestmentAmount: number;
  staticMarketValue: number;  // From database
  liveMarketValue: number;    // From market API
  valuationChange: number;    // Difference
  valuationChangePercent: number;
  tickerSymbol: string;
  currentPrice: number;
  previousClose: number;
  dayChange: number;
  dayChangePercent: number;
  lastUpdated: string;
}

export const useMarketData = (context: BusinessAppContextProps) => {
  const [investments, setInvestments] = useState<EnhancedInvestmentWithMarketData[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const marketDataClient = useMarketDataClient(context.accessToken);
  const { fetchInvestmentsWithTickers } = useGetInvestmentsWithTickers(context);

  const loadMarketData = useCallback(async () => {
    if (!context.accessToken) {
      setError('Access token not available');
      setIsLoading(false);
      return;
    }

    setIsLoading(true);
    setError(null);
    
    try {
      // Step 1: Fetch investments with ticker symbols from database
      const investmentsData = await fetchInvestmentsWithTickers();
      
      // Step 2: Extract ticker symbols
      const tickers = investmentsData
        .map(inv => {
          // Try to get ticker from code field or additionalProperties
          return inv.code || inv.additionalProperties?.tickerSymbol;
        })
        .filter(Boolean);
      
      // Step 3: Fetch real-time market data from external API
      const marketQuotes = await marketDataClient.getBatchQuotes(tickers);
      
      // Step 4: Combine investment data with market data
      const enhancedInvestments = investmentsData.map(investment => {
        const ticker = investment.code || investment.additionalProperties?.tickerSymbol;
        const quote = marketQuotes.quotes[ticker];
        
        if (!quote) {
          // No market data available - use static values
          return {
            id: investment.id,
            code: investment.code,
            name: investment.name,
            investmentType: investment.investmentType?.value,
            totalInvestmentAmount: investment.totalInvestmentAmount,
            staticMarketValue: investment.marketValue,
            liveMarketValue: investment.marketValue,
            valuationChange: 0,
            valuationChangePercent: 0,
            tickerSymbol: ticker,
            currentPrice: 0,
            previousClose: 0,
            dayChange: 0,
            dayChangePercent: 0,
            lastUpdated: new Date().toISOString()
          };
        }
        
        // Calculate live market value (assuming shares count in additionalProperties)
        const shares = investment.additionalProperties?.shares || 1;
        const liveMarketValue = quote.price * shares;
        const valuationChange = liveMarketValue - investment.marketValue;
        const valuationChangePercent = (valuationChange / investment.marketValue) * 100;
        
        return {
          id: investment.id,
          code: investment.code,
          name: investment.name,
          investmentType: investment.investmentType?.value,
          totalInvestmentAmount: investment.totalInvestmentAmount,
          staticMarketValue: investment.marketValue,
          liveMarketValue,
          valuationChange,
          valuationChangePercent,
          tickerSymbol: ticker,
          currentPrice: quote.price,
          previousClose: quote.previousClose,
          dayChange: quote.change,
          dayChangePercent: quote.changePercent,
          lastUpdated: quote.lastUpdated
        };
      });
      
      setInvestments(enhancedInvestments);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Unknown error');
      console.error('Error loading market data:', err);
    } finally {
      setIsLoading(false);
    }
  }, [context.accessToken, fetchInvestmentsWithTickers, marketDataClient]);

  useEffect(() => {
    if (context.accessToken) {
      loadMarketData();
    }
  }, [context.accessToken]); // Only load once on mount

  return {
    investments,
    isLoading,
    error,
    refetch: loadMarketData
  };
};
```
